Ví dụ này là một TLS-enabled shared-nothing worker group có runtime control. Nó
chọn một trong ba listener mode, tạo một TLS context cho mỗi worker và đọc
listener shape, worker count cùng connection capacity từ environment. Module
header mô tả các mode là control cho derived benchmark contract. Chương trình là
`examples/tls_cluster.zig` (219 dòng).

## Source [#source]

Chương trình đầy đủ ở bên dưới. Không phần nào bị lược bỏ.

```zig src/main.zig
//! TLS-enabled shared-nothing worker group: one libxev loop and one TLS
//! context per core.
//!
//! Build and run:
//!   zig build tls_cluster -Doptimize=ReleaseSafe
//!   curl -sk https://127.0.0.1:8443/
//!
//! Optional environment overrides:
//!   UWZ_MODE          tls-cluster (default), plaintext-cluster, or tls-single
//!   UWZ_TLS_BIND      listen address (default 0.0.0.0)
//!   UWZ_TLS_PORT      listen port (default 8443)
//!   UWZ_WORKERS       active worker count (default: platform CPU count;
//!                     UWEBZOCKETS_WORKERS is the canonical spelling and
//!                     UWZ_WORKERS remains a compatibility alias)
//!   UWZ_MAX_CLIENTS   connection capacity: 128, 1024, or 2048 (default 128)
//!   UWZ_CPU_AFFINITY  1 to pin workers to physical cores, 0 to leave them
//!                     unpinned for SMT benchmark hosts (default 1)
//!   UWZ_TLS_CERT      PEM certificate path, paired with UWZ_TLS_KEY
//!   UWZ_TLS_KEY       PEM private key path, paired with UWZ_TLS_CERT
//!
//! The derived benchmark contract uses the three modes as controls. Without
//! credential variables the cluster modes generate one ephemeral self-signed
//! certificate each, which is development-only; the single-application TLS
//! control always requires the PEM files. Each worker terminates TLS on the
//! same SO_REUSEPORT port with its own context, so no handshake state or
//! session cache is shared between loops.

const std = @import("std");
const uz = @import("uWebZockets");

/// Listener shape used by the scaling contract controls.
const Mode = enum {
    tls_cluster,
    plaintext_cluster,
    tls_single,
};

/// Parsed environment overrides for one example run.
const Options = struct {
    mode: Mode = .tls_cluster,
    bind: []const u8 = "0.0.0.0",
    port: u16 = 8443,
    /// Zero selects the CPU-count default or `UWEBZOCKETS_WORKERS`.
    workers: usize = 0,
    max_clients: usize = 128,
    cpu_affinity: bool = true,
    certificate: ?[]const u8 = null,
    private_key: ?[]const u8 = null,
};

fn parse_mode(value: []const u8) !Mode {
    if (std.mem.eql(u8, value, "tls-cluster")) return .tls_cluster;
    if (std.mem.eql(u8, value, "plaintext-cluster")) return .plaintext_cluster;
    if (std.mem.eql(u8, value, "tls-single")) return .tls_single;
    return error.InvalidEnvironmentValue;
}

fn hello(_: *uz.Request, res: *uz.Response) void {
    res.text("hello from a TLS worker group") catch |err| {
        std.log.err("response failed: {}", .{err});
    };
}

fn parse_env_usize(environ: *const std.process.Environ.Map, name: []const u8) !?usize {
    const raw = environ.get(name) orelse return null;
    return std.fmt.parseInt(usize, raw, 10) catch error.InvalidEnvironmentValue;
}

fn read_options(environ: *const std.process.Environ.Map) !Options {
    var options = Options{};
    if (environ.get("UWZ_MODE")) |value| options.mode = try parse_mode(value);
    if (environ.get("UWZ_TLS_BIND")) |value| options.bind = value;
    if (environ.get("UWZ_TLS_PORT")) |value| {
        options.port = std.fmt.parseInt(u16, value, 10) catch return error.InvalidEnvironmentValue;
    }
    if (try parse_env_usize(environ, uz.cluster.workers_environment_variable)) |workers| {
        options.workers = workers;
    } else if (try parse_env_usize(environ, "UWZ_WORKERS")) |workers| {
        options.workers = workers;
    }
    if (try parse_env_usize(environ, "UWZ_MAX_CLIENTS")) |clients| options.max_clients = clients;
    if (environ.get("UWZ_CPU_AFFINITY")) |value| {
        if (std.mem.eql(u8, value, "1")) {
            options.cpu_affinity = true;
        } else if (std.mem.eql(u8, value, "0")) {
            options.cpu_affinity = false;
        } else {
            return error.InvalidEnvironmentValue;
        }
    }
    options.certificate = environ.get("UWZ_TLS_CERT");
    options.private_key = environ.get("UWZ_TLS_KEY");
    if ((options.certificate == null) != (options.private_key == null)) {
        return error.IncompleteTlsCredentials;
    }
    return options;
}

fn log_worker_metrics(worker_index: usize, registry: ?*uz.metrics.Registry) void {
    const counters = registry orelse return;
    std.log.info("worker_accepts index={d} accepted={d} closed={d}", .{
        worker_index,
        counters.get(.connections_accepted),
        counters.get(.connections_closed),
    });
}

fn serve_cluster(
    comptime worker_capacity: usize,
    comptime max_clients: usize,
    workers: usize,
    init: std.process.Init,
    options: Options,
) !void {
    var group = try uz.Server.builder(init.io)
        .with_max_clients(max_clients)
        .with_observability(true)
        .build_cluster_with_worker_count(
        std.heap.page_allocator,
        worker_capacity,
        workers,
        .{ .cpu_affinity = options.cpu_affinity },
    );
    defer group.deinit();

    const Cluster = @TypeOf(group);
    try group.configure(struct {
        fn routes(worker: *Cluster.Worker, worker_index: usize) !void {
            _ = worker_index;
            _ = try worker.get("/", hello);
        }
    }.routes);

    switch (options.mode) {
        .tls_cluster => {
            if (options.certificate) |certificate| {
                const arena = init.arena.allocator();
                const cert_path = try arena.dupeZ(u8, certificate);
                const key_path = try arena.dupeZ(u8, options.private_key.?);
                try group.enable_https(cert_path, key_path);
            } else {
                try group.enable_https_ephemeral();
            }
        },
        .plaintext_cluster => {
            if (options.certificate != null) return error.CredentialsRequireTlsMode;
        },
        .tls_single => return error.WrongListenerMode,
    }

    try group.catch_shutdown_signals();
    try group.listen(options.bind, options.port);
    try group.run();
    for (0..workers) |index| {
        const app_worker = group.worker(index) orelse continue;
        log_worker_metrics(index, app_worker.metrics());
    }
}

fn serve_single(
    comptime max_clients: usize,
    init: std.process.Init,
    options: Options,
) !void {
    const certificate = options.certificate orelse return error.MissingTlsCredentials;
    const private_key = options.private_key orelse return error.MissingTlsCredentials;
    const arena = init.arena.allocator();
    const cert_path = try arena.dupeZ(u8, certificate);
    const key_path = try arena.dupeZ(u8, private_key);

    var application = try uz.App(max_clients).init_https(init.io, cert_path, key_path);
    defer application.deinit();

    _ = try application.get("/", hello);
    try application.catch_shutdown_signals();
    try application.listen(options.bind, options.port);
    try application.run();
}

fn dispatch_clients(
    comptime worker_capacity: usize,
    init: std.process.Init,
    options: Options,
) !void {
    switch (options.max_clients) {
        128 => try dispatch_mode(worker_capacity, 128, init, options),
        256 => try dispatch_mode(worker_capacity, 256, init, options),
        512 => try dispatch_mode(worker_capacity, 512, init, options),
        1024 => try dispatch_mode(worker_capacity, 1024, init, options),
        2048 => try dispatch_mode(worker_capacity, 2048, init, options),
        else => return error.UnsupportedMaxClients,
    }
}

fn dispatch_mode(
    comptime worker_capacity: usize,
    comptime max_clients: usize,
    init: std.process.Init,
    options: Options,
) !void {
    const workers = if (options.workers != 0)
        options.workers
    else if (options.mode == .tls_single)
        1
    else
        try uz.cluster.worker_count_from_environment(init.environ_map, worker_capacity);
    if (workers == 0 or workers > worker_capacity) return error.InvalidWorkerCount;

    if (options.mode == .tls_single) {
        if (workers != 1) return error.SingleModeRequiresOneWorker;
        return serve_single(max_clients, init, options);
    }
    return serve_cluster(worker_capacity, max_clients, workers, init, options);
}

pub fn main(init: std.process.Init) !void {
    const options = try read_options(init.environ_map);
    try dispatch_clients(uz.cluster.default_worker_capacity, init, options);
}
```

## Build wiring [#build-wiring]

Thêm dependency vào project của bạn:

```sh
zig fetch --save 'https://github.com/farbenbuilds/uWebZockets/archive/refs/tags/v2.1.6.tar.gz'
```

Wire module vào `build.zig` như trong
[Run an example in your project](/examples#run-an-example-in-your-project), copy
chương trình trên vào `src/main.zig` và chạy `zig build run`. Build thêm module
import `uWebZockets` và không cần framework checkout.

## Run [#run]

```sh
zig build run
```

Khi không có environment override, process khởi động một worker cho mỗi CPU,
tối đa compiled capacity 256, ở mode `tls-cluster`. Mỗi worker tạo chứng chỉ
self-signed riêng, nên certificate pinning không dùng được trong mode này. Mặc
định này chỉ dành cho development; truyền `UWZ_TLS_CERT` và `UWZ_TLS_KEY` cho
môi trường chia sẻ. Khi truyền cặp này, mỗi worker load nó một lần lúc khởi
động, nên rotation cần rolling restart. Group gọi `catch_shutdown_signals` trước
`listen`, nên Ctrl+C drain các worker.

Trong framework checkout, `zig build tls_cluster -Doptimize=ReleaseSafe` chạy
cùng chương trình. Đó là contributor shortcut, không phải application path.

## Verify [#verify]

```sh
curl -sk https://127.0.0.1:8443/
```

```console
hello from a TLS worker group
```

Body dài 29 byte. `-s` ẩn progress meter, và `-k` bỏ qua xác minh chứng chỉ được
tạo. Với PEM chain mà client tin, dùng `curl -s` không có `-k`.

## Modes [#modes]

| `UWZ_MODE`              | Listener                                                     | Credential                                                                   |
| ----------------------- | ------------------------------------------------------------ | ---------------------------------------------------------------------------- |
| `tls-cluster` (default) | Một TLS listener cho mỗi worker trên một port `SO_REUSEPORT` | `UWZ_TLS_CERT` và `UWZ_TLS_KEY`, hoặc một chứng chỉ ephemeral cho mỗi worker |
| `plaintext-cluster`     | Cùng worker group nhưng không có TLS                         | Phải vắng mặt; cặp credential trả `error.CredentialsRequireTlsMode`          |
| `tls-single`            | Một `App`, một event loop, một TLS context                   | Bắt buộc cặp PEM; thiếu cả cặp trả `error.MissingTlsCredentials`             |

`tls-single` ép active worker count về một. Đặt worker count khác một trả
`error.SingleModeRequiresOneWorker`.

## Environment overrides [#environment-overrides]

| Biến                  | Default               | Giá trị nhận                                     | Tác dụng                                               |
| --------------------- | --------------------- | ------------------------------------------------ | ------------------------------------------------------ |
| `UWZ_MODE`            | `tls-cluster`         | `tls-cluster`, `plaintext-cluster`, `tls-single` | Chọn listener shape                                    |
| `UWZ_TLS_BIND`        | `0.0.0.0`             | Address string                                   | Địa chỉ listen cho mọi listener                        |
| `UWZ_TLS_PORT`        | `8443`                | `u16`                                            | Port listen                                            |
| `UWEBZOCKETS_WORKERS` | CPU count, tối đa 256 | `1` đến compiled capacity                        | Active worker count canonical                          |
| `UWZ_WORKERS`         | Unset                 | `1` đến compiled capacity                        | Alias tương thích, chỉ đọc khi biến canonical vắng mặt |
| `UWZ_MAX_CLIENTS`     | `128`                 | `128`, `256`, `512`, `1024`, `2048`              | Connection capacity mỗi worker                         |
| `UWZ_CPU_AFFINITY`    | `1`                   | `1`, `0`                                         | Pin cluster worker vào physical core                   |
| `UWZ_TLS_CERT`        | Unset                 | PEM path                                         | Certificate chain, đi cặp với `UWZ_TLS_KEY`            |
| `UWZ_TLS_KEY`         | Unset                 | PEM path                                         | Private key, đi cặp với `UWZ_TLS_CERT`                 |

Giá trị mode, port hoặc affinity không hợp lệ trả
`error.InvalidEnvironmentValue`. Chỉ truyền certificate mà thiếu key, hoặc ngược
lại, trả `error.IncompleteTlsCredentials`. Connection capacity dạng số nhưng
ngoài tập giá trị được nhận trả `error.UnsupportedMaxClients`.

Module header liệt kê `UWZ_MAX_CLIENTS` là 128, 1024 hoặc 2048, nhưng
`dispatch_clients` cũng nhận 256 và 512. Bảng dưới đây theo hành vi trong code.

## Key APIs [#key-apis]

| Lệnh gọi                                                              | Tác dụng                                                            |
| --------------------------------------------------------------------- | ------------------------------------------------------------------- |
| `read_options(init.environ_map)`                                      | Parse và validate mọi override tại startup boundary                 |
| `uz.Server.builder(init.io).with_observability(true)`                 | Bật counter registry cho từng worker                                |
| `.build_cluster_with_worker_count(allocator, capacity, workers, .{})` | Compile capacity cố định và khởi động active count chọn lúc runtime |
| `group.configure(...)`                                                | Đăng ký route một lần mỗi worker trước khi thread nào chạy          |
| `group.enable_https(cert_path, key_path)`                             | Load cặp PEM vào một TLS context cho mỗi worker                     |
| `group.enable_https_ephemeral()`                                      | Tạo một chứng chỉ self-signed cho mỗi worker                        |
| `group.catch_shutdown_signals()`                                      | Cài một SIGINT và SIGTERM watcher cho toàn process                  |
| `group.listen(bind, port)`                                            | Bind mọi worker vào cùng port                                       |
| `group.run()`                                                         | Spawn, join và trả về worker failure đầu tiên                       |
| `group.worker(index)`                                                 | Trả một worker, hoặc null khi worker đó không active                |
| `app_worker.metrics()`                                                | Trả counter của worker cho dòng log cuối                            |

## Capacities [#capacities]

| Capacity               |            Value | Set by                          | Consequence                                                |
| ---------------------- | ---------------: | ------------------------------- | ---------------------------------------------------------- |
| Workers                |        CPU count | `worker_count_from_environment` | Tối đa compiled capacity 256                               |
| Connections mỗi worker |              128 | `UWZ_MAX_CLIENTS`               | Tập accepted là 128, 256, 512, 1024, 2048                  |
| Write queue            |           64 KiB | Default                         | Queue đầy trả `error.WouldBlock`                           |
| Body                   |           16 KiB | Default                         | Body lớn hơn nhận `413`                                    |
| Idle timeout           |            120 s | Default                         | Keep-alive connection đã drain sẽ đóng                     |
| HTTP/2 stream          | 8 mỗi connection | Default                         | Stream đồng thời thứ chín nhận `RST_STREAM REFUSED_STREAM` |
| Cluster inbox          |       64 message | Default                         | Inbox đầy trả `error.ClusterQueueFull`                     |
| TLS context            |   Một mỗi worker | `enable_https*`                 | Không handshake state hay session cache nào được chia sẻ   |

## Variation [#variation]

Chạy single-application TLS control với file PEM:

```sh
UWZ_MODE=tls-single \
UWZ_TLS_CERT=certs/fullchain.pem \
UWZ_TLS_KEY=certs/privkey.pem \
zig build run
```

Cách này bỏ worker group: một `App(max_clients)`, một event loop và một TLS
context. Single-application control luôn cần cặp PEM, dù cluster mode có thể
tạo credential ephemeral. `UWZ_CPU_AFFINITY` không có tác dụng trong mode này.
